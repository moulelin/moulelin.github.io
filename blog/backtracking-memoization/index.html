Memoization for Backtracking

June 2, 2023C++LeetcodeBacktracking

Most backtracking involves some repeated searching; a memo can record earlier search results so you can return directly when you reach the same node

idx is an important index: to its left are idx + 1 ~ n-1 (i = idx + 1, idx + 2 … n-1)

Likewise, the values of i in the subtree of idx are also idx + 1 ~ n-1 (i = idx + 1, idx + 2 … n-1)

  • Therefore, using idx and the depth as the memo key is a good way to store this node

Example

link : https://leetcode.com/problems/combinations/description/

class Solution {
public:
     vector<vector<int>> memo[21][21];
    vector<vector<int>> combine(int n, int k, int cur=1) {
        if(k == 0) return {{}};
       
        if(!memo[k][cur].empty()) {
           cout << k << "memo"<<cur;
            return memo[k][cur];
            }
        
        vector<vector<int>> ans;
        for(int i=cur; i<=n; i++){
            cout << "before idx= "<< cur <<" before i=" <<i+1 << " k = "<<k-1<< " ;";
            vector<vector<int>> temp = combine(n, k-1, i+1);
            for(auto v: temp){
                v.push_back(i);
                ans.push_back(v);
            }
            cout << endl;
            cout << "back idx= "<< cur <<" back i="<<i<< " k = "<<k<< endl;
        }
        cout << "end" << endl;
        cout << endl;
        for (auto a:ans){
            cout << "[";
            for (auto b:a) {
                cout << b<<" ";
            }
             cout << "]";
        }
        cout << endl;

        cout << "k=" << k << " idx = " << cur << endl;
        // cout << cur << endl;
        return memo[k][cur] = ans;
    }
};
